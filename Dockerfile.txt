FROM python:3.10-slim

# Install Ghostscript, Potrace, and OpenCV dependencies
RUN apt-get update && apt-get install -y ghostscript potrace libgl1-mesa-glx libglib2.0-0 && rm -rf /var/lib/apt/lists/*

WORKDIR /app
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt
COPY . .

CMD ["sh", "-c", "uvicorn main:app --host 0.0.0.0 --port ${PORT:-10000}"]
